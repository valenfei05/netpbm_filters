cmake_minimum_required(VERSION 3.16)
project(netpbm_filters LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED)
add_library(image src/Image.cpp)
add_library(filters src/filter.cpp src/FilterRunner.cpp)
target_link_libraries(filters PUBLIC image)
add_executable(processor src/processor.cpp)
target_link_libraries(processor PRIVATE image)
add_executable(filterer src/filterer.cpp src/Sequential.cpp)
target_link_libraries(filterer PRIVATE filters)
add_executable(th_filterer src/th_filterer.cpp src/Threaded.cpp)
target_link_libraries(th_filterer PRIVATE filters Threads::Threads)
add_executable(omp_filterer src/omp_filterer.cpp src/OpenMP.cpp)
target_link_libraries(omp_filterer PRIVATE filters OpenMP::OpenMP_CXX)
foreach(target image filters processor filterer th_filterer omp_filterer)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
enable_testing()
add_executable(core_tests tests/core_tests.cpp src/Threaded.cpp src/OpenMP.cpp)
target_link_libraries(core_tests PRIVATE filters Threads::Threads OpenMP::OpenMP_CXX)
add_test(NAME core COMMAND core_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    add_test(NAME correctness COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/validate.py --bin-dir $<TARGET_FILE_DIR:filterer>)
endif()
